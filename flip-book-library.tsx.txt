import DashboardLayout from '@/components/DashboardLayout';
import { Badge } from '@/components/ui/badge';
import { Button } from '@/components/ui/button';
import { Card } from '@/components/ui/card';
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import { useAuth } from '@/contexts/AuthContext';
import { useWorkspace } from '@/contexts/WorkspaceContext';
import { useToast } from '@/hooks/use-toast';
import {
  BookOpen,
  ExternalLink,
  FileText,
  Grid3X3,
  List,
  Link2,
  Loader2,
  Search,
  Trash2,
  Upload,
  X,
} from 'lucide-react';
import { useCallback, useEffect, useMemo, useRef, useState, type KeyboardEvent, type WheelEvent } from 'react';
import HTMLFlipBook from 'react-pageflip';
import { GlobalWorkerOptions, getDocument } from 'pdfjs-dist';
import pdfWorkerUrl from 'pdfjs-dist/build/pdf.worker.min.mjs?url';

// Use a locally bundled worker to avoid external CDN fetch failures.
GlobalWorkerOptions.workerSrc = pdfWorkerUrl;
const FlipBook: any = HTMLFlipBook;

type ViewMode = 'grid' | 'list';

type FlipDocument = {
  id: string;
  title: string;
  description: string;
  createdAt: string;
  fileName: string;
  fileSize: number;
  fileUrl: string;
  pageCount: number;
};

type UploadDraft = {
  title: string;
  description: string;
  file: File | null;
  pageCount: number;
};

function formatFileSize(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
}

export default function FlipBookLibraryPage() {
  const { session } = useAuth();
  const { selectedWorkspace } = useWorkspace();
  const { toast } = useToast();
  const [documents, setDocuments] = useState<FlipDocument[]>([]);
  const [activeDocId, setActiveDocId] = useState<string | null>(null);
  const [loadingDocs, setLoadingDocs] = useState(false);
  const [viewMode, setViewMode] = useState<ViewMode>('grid');
  const [searchQuery, setSearchQuery] = useState('');
  const [uploadOpen, setUploadOpen] = useState(false);
  const [uploading, setUploading] = useState(false);
  const [currentPage, setCurrentPage] = useState(1);
  const [previewImages, setPreviewImages] = useState<string[]>([]);
  const [previewRendering, setPreviewRendering] = useState(false);
  const [previewRenderedPages, setPreviewRenderedPages] = useState(0);
  const [previewZoom, setPreviewZoom] = useState(1);
  const [previewAspectRatio, setPreviewAspectRatio] = useState(1.414);
  const [viewportWidth, setViewportWidth] = useState<number>(
    typeof window === 'undefined' ? 1280 : window.innerWidth,
  );
  const [viewportHeight, setViewportHeight] = useState<number>(
    typeof window === 'undefined' ? 800 : window.innerHeight,
  );
  const [sharingDocId, setSharingDocId] = useState<string | null>(null);
  const [deletingDocId, setDeletingDocId] = useState<string | null>(null);
  const [uploadDraft, setUploadDraft] = useState<UploadDraft>({
    title: '',
    description: '',
    file: null,
    pageCount: 0,
  });

  const flipBookRef = useRef<any>(null);
  const uploadFileInputRef = useRef<HTMLInputElement | null>(null);

  const activeDocument = useMemo(
    () => documents.find((doc) => doc.id === activeDocId) ?? null,
    [documents, activeDocId],
  );

  const flipBookSizeKey = useMemo(
    () => `${activeDocument?.id || 'doc'}-${previewImages.length}`,
    [activeDocument?.id, previewImages.length],
  );

  const totalPreviewPages = previewImages.length || activeDocument?.pageCount || 0;

  const previewPageWidth = useMemo(() => {
    const stageUsableWidth = Math.max(420, viewportWidth * (viewportWidth >= 1024 ? 0.5 : 0.9));
    const stageUsableHeight = Math.max(280, viewportHeight * (viewportWidth >= 1024 ? 0.55 : 0.5));
    const widthBySpread = stageUsableWidth / 2;
    const widthByHeight = stageUsableHeight / Math.max(0.5, previewAspectRatio);
    const nextWidth = Math.floor(Math.min(widthBySpread, widthByHeight));

    if (viewportWidth < 640) return Math.max(210, Math.min(320, nextWidth));
    if (viewportWidth < 1024) return Math.max(260, Math.min(430, nextWidth));
    return Math.max(300, Math.min(620, nextWidth));
  }, [previewAspectRatio, viewportHeight, viewportWidth]);

  const previewPageHeight = useMemo(
    () => Math.max(300, Math.round(previewPageWidth * Math.max(0.5, previewAspectRatio))),
    [previewAspectRatio, previewPageWidth],
  );

  const minPreviewZoom = 0.7;
  const maxPreviewZoom = 2.6;

  const setClampedPreviewZoom = (next: number) => {
    setPreviewZoom(Math.max(minPreviewZoom, Math.min(maxPreviewZoom, Number(next.toFixed(2)))));
  };

  const filteredDocs = useMemo(() => {
    const q = searchQuery.trim().toLowerCase();
    if (!q) return documents;
    return documents.filter((doc) =>
      [doc.title, doc.fileName, doc.description].join(' ').toLowerCase().includes(q),
    );
  }, [documents, searchQuery]);

  const resetUploadDraft = useCallback(() => {
    setUploadDraft({
      title: '',
      description: '',
      file: null,
      pageCount: 0,
    });
  }, []);

  const handleOpenUpload = () => {
    resetUploadDraft();
    setUploadOpen(true);
  };

  const authHeaders = useCallback(() => {
    if (!session?.access_token) return null;
    return {
      Authorization: `Bearer ${session.access_token}`,
    };
  }, [session?.access_token]);

  const fetchDocuments = useCallback(async () => {
    const headers = authHeaders();
    if (!headers) return;

    setLoadingDocs(true);
    try {
      const params = new URLSearchParams();
      if (selectedWorkspace?.id) params.set('workspace_id', selectedWorkspace.id);

      const res = await fetch(`/api/flip-books/documents?${params.toString()}`, {
        headers,
      });

      const json = await res.json().catch(() => ({}));
      if (!res.ok) {
        if (json?.migrationRequired) {
          toast({
            title: 'Migration required',
            description: 'Run backend/migrations/create_flip_books_documents.sql in Supabase SQL Editor.',
            variant: 'destructive',
          });
          return;
        }
        throw new Error(json?.error || 'Failed to load documents');
      }

      const nextDocs: FlipDocument[] = Array.isArray(json?.data) ? json.data : [];
      setDocuments(nextDocs);
      setActiveDocId((prev) => {
        if (prev && nextDocs.some((doc) => doc.id === prev)) return prev;
        return nextDocs[0]?.id ?? null;
      });
    } catch (error: any) {
      toast({ title: 'Failed to load documents', description: error?.message || 'Please try again.', variant: 'destructive' });
    } finally {
      setLoadingDocs(false);
    }
  }, [authHeaders, selectedWorkspace?.id, toast]);

  const handleFileChange = async (file: File | null) => {
    if (!file) {
      setUploadDraft((prev) => ({ ...prev, file: null, pageCount: 0 }));
      return;
    }

    if (file.type !== 'application/pdf') {
      toast({ title: 'Invalid file', description: 'Please select a PDF document only.', variant: 'destructive' });
      return;
    }

    if (file.size > 50 * 1024 * 1024) {
      toast({ title: 'File too large', description: 'Maximum allowed file size is 50MB.', variant: 'destructive' });
      return;
    }

    try {
      const bytes = await file.arrayBuffer();
      const pdf = await getDocument({ data: bytes }).promise;
      setUploadDraft((prev) => ({
        ...prev,
        file,
        pageCount: pdf.numPages,
        title: prev.title || file.name.replace(/\.pdf$/i, ''),
      }));
    } catch {
      // Keep the file selected even if local parsing fails, so server upload can still proceed.
      setUploadDraft((prev) => ({
        ...prev,
        file,
        pageCount: 0,
        title: prev.title || file.name.replace(/\.pdf$/i, ''),
      }));
      toast({
        title: 'Preview unavailable',
        description: 'Could not read page count locally, but upload can still continue.',
        variant: 'default',
      });
    }
  };

  const handleUpload = async () => {
    if (!uploadDraft.file) {
      toast({ title: 'No file selected', description: 'Choose a PDF file first.', variant: 'destructive' });
      return;
    }

    const safeTitle = uploadDraft.title.trim();
    if (!safeTitle) {
      toast({ title: 'Missing title', description: 'Please enter a document title.', variant: 'destructive' });
      return;
    }

    setUploading(true);
    try {
      const headers = authHeaders();
      if (!headers) {
        toast({ title: 'Authentication required', description: 'Please log in again.', variant: 'destructive' });
        return;
      }

      const body = new FormData();
      body.append('pdf', uploadDraft.file);
      body.append('title', safeTitle);
      body.append('description', uploadDraft.description.trim());
      body.append('page_count', String(uploadDraft.pageCount || 1));
      if (selectedWorkspace?.id) {
        body.append('workspace_id', selectedWorkspace.id);
      }

      const res = await fetch('/api/flip-books/documents', {
        method: 'POST',
        headers,
        body,
      });

      const json = await res.json().catch(() => ({}));
      if (!res.ok) {
        if (json?.migrationRequired) {
          toast({
            title: 'Migration required',
            description: 'Run backend/migrations/create_flip_books_documents.sql in Supabase SQL Editor.',
            variant: 'destructive',
          });
          return;
        }
        throw new Error(json?.error || 'Failed to upload document');
      }

      const created = json?.data as FlipDocument | null;
      await fetchDocuments();
      if (created?.id) setActiveDocId(created.id);
      setCurrentPage(1);
      setUploadOpen(false);
      resetUploadDraft();
      toast({ title: 'PDF uploaded', description: 'Your document is ready as an interactive flip book.' });
    } catch (error: any) {
      toast({ title: 'Upload failed', description: error?.message || 'Please try again.', variant: 'destructive' });
    } finally {
      setUploading(false);
    }
  };

  const handleDeleteDoc = async (docId: string) => {
    if (deletingDocId) return;

    const headers = authHeaders();
    if (!headers) return;

    const isActiveDocument = activeDocId === docId;
    const previousActiveDocId = activeDocId;

    setDeletingDocId(docId);

    try {
      // Unmount the active preview before delete to avoid stale node cleanup in the flip engine.
      if (isActiveDocument) {
        setActiveDocId(null);
        setCurrentPage(1);
      }

      const res = await fetch(`/api/flip-books/documents/${docId}`, {
        method: 'DELETE',
        headers,
      });

      if (!res.ok) {
        const json = await res.json().catch(() => ({}));
        if (isActiveDocument) {
          setActiveDocId(previousActiveDocId);
        }
        toast({ title: 'Delete failed', description: json?.error || 'Unable to delete document.', variant: 'destructive' });
        return;
      }

      await fetchDocuments();
      setCurrentPage(1);
    } finally {
      setDeletingDocId(null);
    }
  };

  const clearAll = async () => {
    const headers = authHeaders();
    if (!headers) return;

    const params = new URLSearchParams();
    if (selectedWorkspace?.id) params.set('workspace_id', selectedWorkspace.id);

    const res = await fetch(`/api/flip-books/documents?${params.toString()}`, {
      method: 'DELETE',
      headers,
    });

    if (!res.ok) {
      const json = await res.json().catch(() => ({}));
      toast({ title: 'Clear failed', description: json?.error || 'Unable to clear documents.', variant: 'destructive' });
      return;
    }

    setDocuments([]);
    setActiveDocId(null);
    setCurrentPage(1);
  };

  const flipPrev = () => {
    flipBookRef.current?.pageFlip()?.flipPrev();
  };

  const flipNext = () => {
    flipBookRef.current?.pageFlip()?.flipNext();
  };

  const handlePreviewZoomIn = () => {
    setClampedPreviewZoom(previewZoom + 0.15);
  };

  const handlePreviewZoomOut = () => {
    setClampedPreviewZoom(previewZoom - 0.15);
  };

  const handlePreviewZoomReset = () => {
    setPreviewZoom(1);
  };

  const handlePreviewWheel = (event: WheelEvent<HTMLDivElement>) => {
    event.preventDefault();
    const delta = event.deltaY < 0 ? 0.1 : -0.1;
    setClampedPreviewZoom(previewZoom + delta);
  };

  const handlePreviewKeyDown = (event: KeyboardEvent<HTMLDivElement>) => {
    if (previewRendering) return;

    if (event.key === 'ArrowLeft') {
      event.preventDefault();
      flipPrev();
      return;
    }

    if (event.key === 'ArrowRight') {
      event.preventDefault();
      flipNext();
      return;
    }

    if (event.key === '+' || event.key === '=') {
      event.preventDefault();
      handlePreviewZoomIn();
      return;
    }

    if (event.key === '-' || event.key === '_') {
      event.preventDefault();
      handlePreviewZoomOut();
      return;
    }

    if (event.key === '0') {
      event.preventDefault();
      handlePreviewZoomReset();
    }
  };

  const createShareLink = useCallback(
    async (docId: string) => {
      const headers = authHeaders();
      if (!headers) throw new Error('Authentication required');

      const res = await fetch(`/api/flip-books/documents/${docId}/share-link`, {
        method: 'POST',
        headers,
      });

      const json = await res.json().catch(() => ({}));
      if (!res.ok) {
        throw new Error(json?.error || 'Failed to create share link');
      }

      const shareUrl = String(json?.data?.shareUrl || '').trim();
      if (!shareUrl) throw new Error('Share URL was empty');
      return shareUrl;
    },
    [authHeaders],
  );

  const handleOpenInWindow = useCallback(async () => {
    if (!activeDocument?.id) return;
    setSharingDocId(activeDocument.id);
    try {
      const shareUrl = await createShareLink(activeDocument.id);
      window.open(shareUrl, '_blank', 'noopener,noreferrer');
    } catch (error: any) {
      toast({
        title: 'Unable to open window',
        description: error?.message || 'Please try again.',
        variant: 'destructive',
      });
    } finally {
      setSharingDocId(null);
    }
  }, [activeDocument?.id, createShareLink, toast]);

  const handleCopyShareLink = useCallback(async () => {
    if (!activeDocument?.id) return;
    setSharingDocId(activeDocument.id);
    try {
      const shareUrl = await createShareLink(activeDocument.id);
      await navigator.clipboard.writeText(shareUrl);
      toast({ title: 'Share link copied', description: 'You can now share this flip book link.' });
    } catch (error: any) {
      toast({
        title: 'Unable to copy link',
        description: error?.message || 'Please try again.',
        variant: 'destructive',
      });
    } finally {
      setSharingDocId(null);
    }
  }, [activeDocument?.id, createShareLink, toast]);

  useEffect(() => {
    if (!session?.access_token) return;
    fetchDocuments();
  }, [session?.access_token, selectedWorkspace?.id, fetchDocuments]);

  useEffect(() => {
    setCurrentPage(1);
    setPreviewZoom(1);
  }, [activeDocId]);

  useEffect(() => {
    if (typeof window === 'undefined') return;

    const onResize = () => {
      setViewportWidth(window.innerWidth);
      setViewportHeight(window.innerHeight);
    };

    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, []);

  useEffect(() => {
    if (!activeDocument?.id) {
      setPreviewImages([]);
      setPreviewRenderedPages(0);
      setPreviewRendering(false);
      return;
    }

    const headers = authHeaders();
    if (!headers) {
      setPreviewImages([]);
      setPreviewRenderedPages(0);
      setPreviewRendering(false);
      return;
    }

    let cancelled = false;
    const task = (async () => {
      try {
        setPreviewRendering(true);
        setPreviewRenderedPages(0);
        setCurrentPage(1);
        setPreviewImages([]);

        const response = await fetch(`/api/flip-books/documents/${activeDocument.id}/file`, {
          headers,
        });

        if (!response.ok) {
          throw new Error('Unable to download PDF for preview.');
        }

        const pdfBytes = await response.arrayBuffer();
        const pdf = await getDocument({ data: pdfBytes }).promise;
        const urls: string[] = [];

        for (let pageNumber = 1; pageNumber <= pdf.numPages; pageNumber += 1) {
          if (cancelled) break;

          const page = await pdf.getPage(pageNumber);
          const viewport = page.getViewport({ scale: 1 });
          if (pageNumber === 1 && viewport.width > 0 && viewport.height > 0) {
            setPreviewAspectRatio(viewport.height / viewport.width);
          }

          const targetWidth = 1200;
          const scale = Math.min(2, Math.max(1, targetWidth / Math.max(1, viewport.width)));
          const renderViewport = page.getViewport({ scale });

          const canvas = document.createElement('canvas');
          canvas.width = Math.max(1, Math.floor(renderViewport.width));
          canvas.height = Math.max(1, Math.floor(renderViewport.height));

          const context = canvas.getContext('2d', { alpha: false });
          if (!context) throw new Error('Unable to initialize preview renderer.');

          await page.render({ canvas, canvasContext: context, viewport: renderViewport }).promise;
          urls.push(canvas.toDataURL('image/jpeg', 0.92));
          setPreviewRenderedPages(pageNumber);
          canvas.width = 0;
          canvas.height = 0;
        }

        if (cancelled) return;
        if (!urls.length) throw new Error('No pages were rendered for preview.');

        setPreviewImages(urls);
        setCurrentPage(1);
      } catch (error: any) {
        if (cancelled) return;
        toast({
          title: 'Preview failed',
          description: error?.message || 'Could not render this PDF preview.',
          variant: 'destructive',
        });
      } finally {
        if (!cancelled) setPreviewRendering(false);
      }
    })();

    return () => {
      cancelled = true;
      void task;
    };
  }, [activeDocument?.id, authHeaders, toast]);

  useEffect(() => {
    if (!totalPreviewPages) {
      setCurrentPage(1);
      return;
    }

    setCurrentPage((prev) => Math.max(1, Math.min(prev, totalPreviewPages)));
  }, [totalPreviewPages]);

  return (
    <DashboardLayout>
      <div className="space-y-6 pb-4 md:pb-6">
        <div className="rounded-3xl border border-slate-200 bg-white p-6 shadow-sm">
          <div className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">
            <div>
              <h1 className="text-2xl font-semibold text-slate-900">Flip Book Library</h1>
              <p className="mt-1 text-sm text-slate-500">Upload PDF documents and read them with realistic page turns.</p>
            </div>
            <div className="flex items-center gap-2">
              <Button variant="outline" className="gap-2" onClick={clearAll} disabled={!documents.length}>
                <Trash2 className="h-4 w-4" />
                Clear All
              </Button>
              <Button className="gap-2 bg-[linear-gradient(135deg,#1C7ED6_0%,#37B24D_100%)] text-white" onClick={handleOpenUpload}>
                <Upload className="h-4 w-4" />
                Upload PDF
              </Button>
            </div>
          </div>

          <div className="mt-5 flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
            <div className="relative w-full md:max-w-md">
              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
              <Input
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                placeholder="Search documents..."
                className="pl-9"
              />
            </div>

            <div className="inline-flex rounded-xl border border-slate-200 bg-slate-50 p-1">
              <Button
                variant={viewMode === 'grid' ? 'default' : 'ghost'}
                size="sm"
                className="gap-2"
                onClick={() => setViewMode('grid')}
              >
                <Grid3X3 className="h-4 w-4" /> Grid
              </Button>
              <Button
                variant={viewMode === 'list' ? 'default' : 'ghost'}
                size="sm"
                className="gap-2"
                onClick={() => setViewMode('list')}
              >
                <List className="h-4 w-4" /> List
              </Button>
            </div>
          </div>
        </div>

        <div className="grid gap-6 lg:grid-cols-[360px,1fr]">
          <Card className="rounded-3xl border-slate-200 bg-white p-4">
            <div className="mb-3 flex items-center justify-between px-1">
              <h2 className="text-sm font-semibold text-slate-800">Documents</h2>
              <Badge variant="secondary">{filteredDocs.length}</Badge>
            </div>

            <div className="max-h-[70vh] space-y-2 overflow-y-auto pr-1">
              {loadingDocs && (
                <div className="rounded-2xl border border-slate-200 bg-slate-50 p-4 text-center text-sm text-slate-500">
                  Loading documents...
                </div>
              )}

              {!filteredDocs.length && (
                <div className="rounded-2xl border border-dashed border-slate-200 bg-slate-50 p-6 text-center">
                  <BookOpen className="mx-auto h-8 w-8 text-slate-400" />
                  <p className="mt-2 text-sm text-slate-500">No documents found.</p>
                </div>
              )}

              {filteredDocs.map((doc) => {
                const selected = doc.id === activeDocId;
                return (
                  <div
                    key={doc.id}
                    role="button"
                    tabIndex={0}
                    onClick={() => {
                      setActiveDocId(doc.id);
                      setCurrentPage(1);
                    }}
                    onKeyDown={(e) => {
                      if (e.key === 'Enter' || e.key === ' ') {
                        e.preventDefault();
                        setActiveDocId(doc.id);
                        setCurrentPage(1);
                      }
                    }}
                    className={`w-full rounded-2xl border p-3 text-left transition ${
                      selected
                        ? 'border-[#1C7ED6] bg-[#EFF6FF]'
                        : 'border-slate-200 bg-white hover:border-slate-300 hover:bg-slate-50'
                    }`}
                  >
                    <div className="flex items-start justify-between gap-3">
                      <div className="min-w-0">
                        <p className="truncate text-sm font-semibold text-slate-800">{doc.title}</p>
                        <p className="mt-0.5 truncate text-xs text-slate-500">{doc.fileName}</p>
                        <div className="mt-2 flex items-center gap-2 text-[11px] text-slate-500">
                          <span>{doc.pageCount} pages</span>
                          <span>•</span>
                          <span>{formatFileSize(doc.fileSize)}</span>
                        </div>
                      </div>
                      <span
                        role="button"
                        tabIndex={0}
                        onClick={(e) => {
                          e.stopPropagation();
                          if (deletingDocId) return;
                          handleDeleteDoc(doc.id);
                        }}
                        onKeyDown={(e) => {
                          if (e.key === 'Enter' || e.key === ' ') {
                            e.preventDefault();
                            e.stopPropagation();
                            if (deletingDocId) return;
                            handleDeleteDoc(doc.id);
                          }
                        }}
                        className="rounded-md p-1 text-slate-400 hover:bg-slate-100 hover:text-slate-600 disabled:pointer-events-none disabled:opacity-50"
                        aria-label="Delete document"
                        aria-disabled={Boolean(deletingDocId)}
                      >
                        {deletingDocId === doc.id ? (
                          <span className="h-4 w-4 animate-pulse text-[10px] font-semibold">...</span>
                        ) : (
                          <X className="h-4 w-4" />
                        )}
                      </span>
                    </div>
                  </div>
                );
              })}
            </div>
          </Card>

          <Card className="relative overflow-hidden rounded-3xl border-slate-200 bg-[radial-gradient(ellipse_at_top,#0b1f48_0%,#071735_42%,#06132c_100%)] p-4 text-white">
            {!activeDocument ? (
              <div className="flex min-h-[58vh] flex-col items-center justify-center py-8 text-center md:min-h-[64vh] lg:min-h-[68vh]">
                <div className="rounded-full bg-white/10 p-6">
                  <BookOpen className="h-12 w-12 text-white/70" />
                </div>
                <h3 className="mt-5 text-3xl font-bold">No Documents Yet</h3>
                <p className="mt-2 max-w-md text-sm text-slate-300">
                  Upload a PDF to preview it as an interactive flip book with page-turn controls.
                </p>
                <Button
                  onClick={handleOpenUpload}
                  className="mt-6 bg-[linear-gradient(135deg,#6EA8FE_0%,#37B24D_100%)] text-slate-950 hover:brightness-110"
                >
                  Choose PDF
                </Button>
              </div>
            ) : (
              <>
                <div className="mb-4 rounded-xl bg-white/10 p-3 backdrop-blur-sm">
                  <div className="flex flex-wrap items-center justify-between gap-3">
                    <div>
                      <p className="text-sm font-semibold">{activeDocument.title}</p>
                      <p className="text-xs text-slate-300">
                        {activeDocument.pageCount} pages • {formatFileSize(activeDocument.fileSize)}
                      </p>
                    </div>
                    <div className="flex flex-wrap items-center justify-end gap-2">
                      <Button
                        variant="secondary"
                        size="sm"
                        className="gap-1"
                        onClick={handleOpenInWindow}
                        disabled={sharingDocId === activeDocument.id}
                      >
                        <ExternalLink className="h-4 w-4" /> Open Window
                      </Button>
                      <Button
                        variant="secondary"
                        size="sm"
                        className="gap-1"
                        onClick={handleCopyShareLink}
                        disabled={sharingDocId === activeDocument.id}
                      >
                        <Link2 className="h-4 w-4" /> Share Link
                      </Button>
                      <Badge className="bg-white/20 text-white">Page {currentPage} / {totalPreviewPages || activeDocument.pageCount}</Badge>
                    </div>
                  </div>
                </div>

                <div
                  className="flex h-[64vh] items-center justify-center overflow-hidden rounded-2xl border border-white/10 bg-[#000000] outline-none"
                  onWheel={handlePreviewWheel}
                  onKeyDown={handlePreviewKeyDown}
                  tabIndex={0}
                >
                  {previewRendering ? (
                    <div className="flex flex-col items-center gap-3 text-sm text-slate-300">
                      <Loader2 className="h-5 w-5 animate-spin" />
                      <p>Rendering pages {Math.max(0, previewRenderedPages)} / {activeDocument.pageCount}</p>
                      <p className="text-xs text-slate-500">Use mouse wheel to zoom</p>
                    </div>
                  ) : previewImages.length ? (
                    <div className="flex items-center justify-center transition-transform duration-150" style={{ transform: `scale(${previewZoom})` }}>
                      <FlipBook
                        key={flipBookSizeKey}
                        width={previewPageWidth}
                        height={previewPageHeight}
                        size="fixed"
                        minWidth={Math.max(220, Math.floor(previewPageWidth * 0.78))}
                        maxWidth={Math.max(previewPageWidth, 1200)}
                        minHeight={Math.max(280, Math.floor(previewPageHeight * 0.78))}
                        maxHeight={Math.max(previewPageHeight, 1700)}
                        drawShadow={true}
                        flippingTime={800}
                        usePortrait={false}
                        startZIndex={0}
                        autoSize={true}
                        clickEventForward={true}
                        useMouseEvents={true}
                        swipeDistance={30}
                        showPageCorners={true}
                        disableFlipByClick={false}
                        ref={flipBookRef}
                        onFlip={(e: any) => {
                          setCurrentPage((Number(e?.data) || 0) + 1);
                        }}
                      >
                        {previewImages.map((pageImageUrl, index) => (
                          <div
                            key={`${activeDocument.id}-${index + 1}`}
                            className="h-full w-full overflow-hidden bg-black"
                            style={{ border: '1px solid rgba(255,255,255,0.06)' }}
                          >
                            <img
                              src={pageImageUrl}
                              alt={`Preview page ${index + 1}`}
                              style={{ width: '100%', height: '100%', objectFit: 'contain' }}
                              draggable={false}
                            />
                          </div>
                        ))}
                      </FlipBook>
                    </div>
                  ) : (
                    <div className="text-sm text-slate-400">No pages available for preview.</div>
                  )}
                </div>
              </>
            )}
          </Card>
        </div>
      </div>

      <Dialog open={uploadOpen} onOpenChange={setUploadOpen}>
        <DialogContent className="max-w-2xl border-slate-200 bg-white">
          <DialogHeader>
            <DialogTitle className="text-xl text-slate-900">Upload PDF</DialogTitle>
            <DialogDescription>
              Upload a PDF file to add it to your flip book library.
            </DialogDescription>
          </DialogHeader>

          <div className="space-y-4">
            <div className="space-y-2">
              <label className="text-sm font-medium text-slate-700">Select PDF File</label>
              <input
                ref={uploadFileInputRef}
                type="file"
                accept="application/pdf"
                className="hidden"
                onChange={(e) => handleFileChange(e.target.files?.[0] || null)}
              />
              <div className="flex items-center overflow-hidden rounded-md border border-slate-200 bg-white">
                <div className="flex-1 truncate px-3 py-2 text-sm text-slate-500">
                  {uploadDraft.file?.name || 'No file selected'}
                </div>
                <Button
                  type="button"
                  variant="outline"
                  className="h-full rounded-none border-y-0 border-r-0 border-l-slate-200 bg-[linear-gradient(135deg,#1C7ED6_0%,#37B24D_100%)] px-4 font-semibold text-white hover:brightness-110"
                  onClick={() => uploadFileInputRef.current?.click()}
                >
                  Choose File
                </Button>
              </div>
              <p className="text-xs text-slate-500">PDF only, max 50MB.</p>
            </div>

            <div className="space-y-2">
              <label className="text-sm font-medium text-slate-700">Document Title</label>
              <Input
                value={uploadDraft.title}
                onChange={(e) => setUploadDraft((prev) => ({ ...prev, title: e.target.value }))}
                placeholder="Enter document title"
                maxLength={120}
              />
            </div>

            <div className="space-y-2">
              <label className="text-sm font-medium text-slate-700">Description (Optional)</label>
              <Textarea
                value={uploadDraft.description}
                onChange={(e) => setUploadDraft((prev) => ({ ...prev, description: e.target.value }))}
                placeholder="Enter document description"
                maxLength={500}
              />
            </div>

            {uploadDraft.file && (
              <div className="rounded-xl border border-slate-200 bg-slate-50 p-3 text-xs text-slate-600">
                <div className="flex items-center gap-2">
                  <FileText className="h-4 w-4" />
                  <span className="font-medium">{uploadDraft.file.name}</span>
                </div>
                <p className="mt-1">{uploadDraft.pageCount} pages • {formatFileSize(uploadDraft.file.size)}</p>
              </div>
            )}
          </div>

          <DialogFooter>
            <Button variant="ghost" onClick={() => setUploadOpen(false)} disabled={uploading}>
              Cancel
            </Button>
            <Button
              onClick={handleUpload}
              disabled={uploading || !uploadDraft.file || !uploadDraft.title.trim()}
              className="bg-[linear-gradient(135deg,#1C7ED6_0%,#37B24D_100%)] text-white"
            >
              {uploading ? 'Uploading...' : 'Upload & Convert'}
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </DashboardLayout>
  );
}
