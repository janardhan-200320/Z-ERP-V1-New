import { Badge } from '@/components/ui/badge';
import { Button } from '@/components/ui/button';
import { useToast } from '@/hooks/use-toast';
import {
  ChevronLeft,
  Link2,
  Loader2,
  ChevronRight,
} from 'lucide-react';
import { forwardRef, useCallback, useEffect, useMemo, useRef, useState } from 'react';
import HTMLFlipBook from 'react-pageflip';
import { Document, Page, pdfjs } from 'react-pdf';
import pdfWorkerUrl from 'pdfjs-dist/build/pdf.worker.min.mjs?url';
import 'react-pdf/dist/Page/AnnotationLayer.css';
import 'react-pdf/dist/Page/TextLayer.css';
import { useRoute } from 'wouter';

pdfjs.GlobalWorkerOptions.workerSrc = pdfWorkerUrl;
const FlipBook: any = HTMLFlipBook;

type FlipDirection = 'forward' | 'backward';

type SharedFlipDocument = {
  id: string;
  title: string;
  description: string;
  fileName: string;
  fileSize: number;
  fileUrl: string;
  pageCount: number;
  createdAt: string;
};

type FlipPageProps = {
  pageNumber: number;
  pageWidth: number;
  side: 'left' | 'right';
  onCanvasMeasure?: (width: number, height: number) => void;
};

const FlipPage = forwardRef<HTMLDivElement, FlipPageProps>(function FlipPage(
  { pageNumber, pageWidth, side, onCanvasMeasure },
  ref,
) {
  return (
    <div ref={ref} className={`book-sheet ${side === 'left' ? 'is-left' : 'is-right'}`}>
      <div className="book-sheet__paper">
        <div className="book-sheet__curl book-sheet__curl--near" aria-hidden="true" />
        <div className="book-sheet__curl book-sheet__curl--mid" aria-hidden="true" />
        <div className="book-sheet__curl book-sheet__curl--far" aria-hidden="true" />
        <div className="book-sheet__edge" aria-hidden="true" />
        <Page
          pageNumber={pageNumber}
          width={pageWidth}
          renderMode="canvas"
          canvasBackground="#ffffff"
          className="book-sheet__pdf-page"
          canvasRef={(canvas) => {
            if (!canvas) return;
            if (canvas.width > 0 && canvas.height > 0) {
              onCanvasMeasure?.(canvas.width, canvas.height);
            }
          }}
          renderTextLayer={false}
          renderAnnotationLayer={false}
          loading={<div className="p-8 text-sm text-slate-500">Loading page...</div>}
        />
      </div>
    </div>
  );
});

export default function FlipBookViewerPage() {
  const [, params] = useRoute('/flip-books/view/:id');
  const { toast } = useToast();

  const [loading, setLoading] = useState(true);
  const [documentData, setDocumentData] = useState<SharedFlipDocument | null>(null);
  const [currentPage, setCurrentPage] = useState(1);
  const [flipDirection, setFlipDirection] = useState<FlipDirection>('forward');
  const [flipDuration, setFlipDuration] = useState(980);
  const [isAnimating, setIsAnimating] = useState(false);
  const [flipProgress, setFlipProgress] = useState(0);
  const [pageAspectRatio, setPageAspectRatio] = useState(1 / 1.414);
  const [layoutReady, setLayoutReady] = useState(false);
  const [viewportWidth, setViewportWidth] = useState<number>(
    typeof window === 'undefined' ? 1280 : window.innerWidth,
  );
  const [viewportHeight, setViewportHeight] = useState<number>(
    typeof window === 'undefined' ? 800 : window.innerHeight,
  );

  const flipBookRef = useRef<any>(null);
  const lastPageIndexRef = useRef(0);
  const flipStartRef = useRef<number | null>(null);
  const rafRef = useRef<number | null>(null);
  const aspectLockedRef = useRef(false);
  const flipSafetyTimerRef = useRef<number | null>(null);

  const token = useMemo(() => {
    if (typeof window === 'undefined') return '';
    return new URLSearchParams(window.location.search).get('token') || '';
  }, []);

  const safePageAspectRatio = useMemo(() => {
    if (!Number.isFinite(pageAspectRatio) || pageAspectRatio <= 0) return 1 / 1.414;
    return Math.min(3, Math.max(0.35, pageAspectRatio));
  }, [pageAspectRatio]);

  const pageWidth = useMemo(() => {
    const spreadFraction = viewportWidth < 640 ? 0.94 : 0.92;
    const stageUsableWidth = Math.max(320, viewportWidth * spreadFraction);
    const widthBySpread = stageUsableWidth / 2;

    const stageUsableHeight = Math.max(180, viewportHeight * 0.78);
    const widthByHeight = stageUsableHeight * safePageAspectRatio;

    const rawWidth = Math.floor(Math.min(widthBySpread, widthByHeight));

    if (viewportWidth < 640) return Math.min(320, Math.max(145, rawWidth));
    if (viewportWidth < 1024) return Math.min(480, Math.max(220, rawWidth));
    if (viewportWidth < 1440) return Math.min(640, Math.max(300, rawWidth));
    return Math.min(760, Math.max(360, rawWidth));
  }, [safePageAspectRatio, viewportWidth, viewportHeight]);

  const pageHeight = useMemo(
    () => Math.max(100, Math.round(pageWidth / safePageAspectRatio)),
    [pageWidth, safePageAspectRatio],
  );

  const maxBookWidth = useMemo(() => pageWidth + 8, [pageWidth]);

  const maxBookHeight = useMemo(() => pageHeight + 8, [pageHeight]);

  const flipBookSizeKey = useMemo(
    () => `${documentData?.id || 'doc'}`,
    [documentData?.id],
  );
  const isDesktop = viewportWidth >= 1024;

  const canFlipPrev = currentPage > 1;
  const canFlipNext = currentPage < (documentData?.pageCount || 0);

  const onFlipStart = () => {
    if (flipSafetyTimerRef.current) {
      window.clearTimeout(flipSafetyTimerRef.current);
      flipSafetyTimerRef.current = null;
    }

    setIsAnimating(true);
    setFlipProgress(0);
    flipStartRef.current = null;

    flipSafetyTimerRef.current = window.setTimeout(() => {
      setIsAnimating(false);
      setFlipProgress(0);
      flipStartRef.current = null;
      flipSafetyTimerRef.current = null;
    }, Math.max(600, flipDuration + 260));
  };

  const onFlipEnd = () => {
    if (flipSafetyTimerRef.current) {
      window.clearTimeout(flipSafetyTimerRef.current);
      flipSafetyTimerRef.current = null;
    }

    setIsAnimating(false);
    setFlipProgress(1);
    window.setTimeout(() => setFlipProgress(0), 80);
  };

  const handlePdfLoadSuccess = useCallback(async (pdf: any) => {
    try {
      const samples = Math.max(1, Math.min(4, Number(pdf?.numPages || 1)));
      let ratio = 1 / 1.414;

      for (let i = 1; i <= samples; i += 1) {
        const page = await pdf.getPage(i);
        const viewport = page.getViewport({ scale: 1 });
        const width = Number(viewport?.width || 0);
        const height = Number(viewport?.height || 0);
        if (width > 0 && height > 0) {
          ratio = Math.max(ratio, width / height);
        }
      }

      setPageAspectRatio(Math.min(3, Math.max(0.35, ratio)));
      aspectLockedRef.current = true;
      setLayoutReady(true);
    } catch {
      setPageAspectRatio(1 / 1.414);
      aspectLockedRef.current = true;
      setLayoutReady(true);
    }
  }, []);

  const handleCanvasMeasure = useCallback((width: number, height: number) => {
    if (aspectLockedRef.current) return;
    if (width <= 0 || height <= 0) return;
    const ratio = width / height;
    if (!Number.isFinite(ratio)) return;
    const normalized = Math.min(3, Math.max(0.35, ratio));
    aspectLockedRef.current = true;
    setPageAspectRatio((prev) => (Math.abs(prev - normalized) > 0.02 ? normalized : prev));
  }, []);

  useEffect(() => {
    const documentId = params?.id;
    if (!documentId || !token) {
      setLoading(false);
      return;
    }

    let active = true;
    setLoading(true);

    fetch(`/api/flip-books/public/${documentId}?token=${encodeURIComponent(token)}`)
      .then(async (res) => {
        const json = await res.json().catch(() => ({}));
        if (!res.ok) throw new Error(json?.error || 'Unable to open shared flip book');

        if (active) {
          setDocumentData(json?.data || null);
          setCurrentPage(1);
          setFlipDirection('forward');
          setFlipDuration(isDesktop ? 760 : 680);
          setPageAspectRatio(1 / 1.414);
          setLayoutReady(false);
          aspectLockedRef.current = false;
          onFlipEnd();
          lastPageIndexRef.current = 0;
        }
      })
      .catch((error: any) => {
        toast({
          title: 'Shared link unavailable',
          description: error?.message || 'Invalid or expired link.',
          variant: 'destructive',
        });
      })
      .finally(() => {
        if (active) setLoading(false);
      });

    return () => {
      active = false;
    };
  }, [params?.id, token, toast, isDesktop]);

  useEffect(() => {
    if (typeof window === 'undefined') return;

    const onResize = () => {
      setViewportWidth(window.innerWidth);
      setViewportHeight(window.innerHeight);
    };
    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, []);

  useEffect(() => {
    return () => {
      if (flipSafetyTimerRef.current) {
        window.clearTimeout(flipSafetyTimerRef.current);
        flipSafetyTimerRef.current = null;
      }
    };
  }, []);

  useEffect(() => {
    if (!isAnimating) {
      flipStartRef.current = null;
      if (rafRef.current) {
        cancelAnimationFrame(rafRef.current);
        rafRef.current = null;
      }
      setFlipProgress(0);
      return;
    }

    const animate = (now: number) => {
      if (flipStartRef.current == null) flipStartRef.current = now;
      const elapsed = now - flipStartRef.current;
      const progress = Math.min(1, elapsed / Math.max(1, flipDuration));
      setFlipProgress(progress);
      if (progress < 1) {
        rafRef.current = requestAnimationFrame(animate);
      }
    };

    rafRef.current = requestAnimationFrame(animate);
    return () => {
      if (rafRef.current) {
        cancelAnimationFrame(rafRef.current);
        rafRef.current = null;
      }
    };
  }, [isAnimating, flipDuration]);

  const flipPrev = () => {
    if (isAnimating) return;
    setFlipDirection('backward');
    setFlipDuration(isDesktop ? 620 : 560);
    flipBookRef.current?.pageFlip()?.flipPrev();
  };

  const flipNext = () => {
    if (isAnimating) return;
    setFlipDirection('forward');
    setFlipDuration(isDesktop ? 760 : 680);
    flipBookRef.current?.pageFlip()?.flipNext();
  };

  if (loading) {
    return (
      <div className="flex min-h-screen items-center justify-center bg-black text-white">
        <div className="flex items-center gap-2 text-sm text-slate-200">
          <Loader2 className="h-4 w-4 animate-spin" /> Preparing flip book...
        </div>
      </div>
    );
  }

  if (!documentData) {
    return (
      <div className="flex min-h-screen items-center justify-center bg-black p-6 text-center text-white">
        <div>
          <h1 className="text-2xl font-semibold">This flip book is unavailable</h1>
          <p className="mt-2 text-sm text-slate-300">The link is invalid, expired, or the document was removed.</p>
        </div>
      </div>
    );
  }

  return (
    <div className="h-screen bg-black text-white">
      <div className="relative mx-auto flex h-screen max-w-[1800px] flex-col">
        <div className="z-20 flex flex-wrap items-center justify-between gap-3 border-b border-white/10 bg-black/65 px-4 py-3 backdrop-blur-sm md:px-6">
          <div>
            <h1 className="text-lg font-semibold md:text-xl">{documentData.title}</h1>
            <p className="text-xs text-slate-300 md:text-sm">{documentData.pageCount} pages</p>
          </div>

          <div className="flex items-center gap-2">
            <Button
              variant="secondary"
              size="sm"
              className="gap-1"
              onClick={async () => {
                await navigator.clipboard.writeText(window.location.href);
                toast({ title: 'Link copied', description: 'Share link copied to clipboard.' });
              }}
            >
              <Link2 className="h-4 w-4" /> Copy Link
            </Button>
            <Badge className="bg-white/20 text-white">Page {currentPage} / {documentData.pageCount}</Badge>
          </div>
        </div>

        <div className="relative flex flex-1 items-center justify-center p-4 md:p-8">
          <button
            type="button"
            aria-label="Previous spread"
            onClick={flipPrev}
            disabled={!canFlipPrev || isAnimating}
            className="absolute left-2 top-1/2 z-30 -translate-y-1/2 rounded-full p-2 text-white/70 transition hover:text-white disabled:cursor-not-allowed disabled:opacity-30 md:left-6"
          >
            <ChevronLeft className="h-8 w-8" />
          </button>

          <button
            type="button"
            aria-label="Next spread"
            onClick={flipNext}
            disabled={!canFlipNext || isAnimating}
            className="absolute right-2 top-1/2 z-30 -translate-y-1/2 rounded-full p-2 text-white/70 transition hover:text-white disabled:cursor-not-allowed disabled:opacity-30 md:right-6"
          >
            <ChevronRight className="h-8 w-8" />
          </button>

          <div
            className={`book-stage shared-viewer-stage flex h-[84vh] w-full items-center justify-center overflow-hidden rounded-2xl border border-white/10 bg-black/90 ${flipDirection === 'forward' ? 'is-forward' : 'is-backward'} ${isAnimating ? 'is-flipping' : ''}`}
            style={{
              ['--flip-duration' as any]: `${flipDuration}ms`,
              ['--flip-progress' as any]: String(flipProgress),
            }}
          >
            <div className="book-stage__glow" />
            <div className="book-stage__surface book-stage__surface--large">
              <div className="book-spine" aria-hidden="true" />
              <div className="magazine-book">
                <Document
                  file={documentData.fileUrl}
                  loading={<div className="text-sm text-slate-300">Loading PDF...</div>}
                  onLoadSuccess={handlePdfLoadSuccess}
                  onLoadError={(error) => {
                    setLayoutReady(true);
                    toast({
                      title: 'Preview failed',
                      description: (error as Error)?.message || 'Could not render this PDF.',
                      variant: 'destructive',
                    });
                  }}
                >
                  {layoutReady ? (
                    <FlipBook
                      key={flipBookSizeKey}
                      width={pageWidth}
                      height={pageHeight}
                      size="fixed"
                      minWidth={130}
                      maxWidth={maxBookWidth}
                      minHeight={Math.max(100, Math.floor(pageHeight * 0.7))}
                      maxHeight={maxBookHeight}
                      maxShadowOpacity={0.62}
                      mobileScrollSupport={true}
                      className="magazine-book__engine"
                      startPage={0}
                      drawShadow={false}
                      flippingTime={flipDuration}
                      usePortrait={false}
                      startZIndex={1}
                      showPageCorners={false}
                      showCover={false}
                      ref={flipBookRef}
                      onChangeState={(e: any) => {
                        const state = String(e?.data || '').toLowerCase();
                        if (state === 'flipping') {
                          onFlipStart();
                          return;
                        }
                        onFlipEnd();
                      }}
                      onFlip={(e: any) => {
                        const pageIndex = Number(e.data) || 0;
                        const prev = lastPageIndexRef.current;
                        const nextDirection: FlipDirection = pageIndex >= prev ? 'forward' : 'backward';
                        setFlipDirection(nextDirection);
                        setFlipDuration(isDesktop ? (nextDirection === 'forward' ? 760 : 620) : 640);
                        setCurrentPage(pageIndex + 1);
                        lastPageIndexRef.current = pageIndex;
                        onFlipEnd();
                      }}
                    >
                      {Array.from({ length: documentData.pageCount }).map((_, index) => (
                        <FlipPage
                          key={`${documentData.id}-${index + 1}`}
                          pageNumber={index + 1}
                          pageWidth={pageWidth - 14}
                          side={(index + 1) % 2 === 0 ? 'left' : 'right'}
                          onCanvasMeasure={handleCanvasMeasure}
                        />
                      ))}
                    </FlipBook>
                  ) : (
                    <div className="flex items-center justify-center rounded-md border border-white/10 bg-black/30 px-6 py-3 text-sm text-slate-300">
                      Preparing page layout...
                    </div>
                  )}
                </Document>
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
